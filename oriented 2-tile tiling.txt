% %% ================= 辅助函数 (完整恢复) =================
function p = integer_partitions(n, k, min_val)
    if nargin < 3, min_val = 1; end
    if k == 0
        if n == 0, p = {[]}; else, p = {}; end
        return;
    end
    if n < k * min_val
        p = {};
        return;
    end
    p = {};
    for i = min_val : floor(n / k)
        sub_partitions = integer_partitions(n - i, k - 1, i);
        for j = 1:length(sub_partitions)
            p{end+1} = [i, sub_partitions{j}];
        end
    end
end


function vertex_cells = get_vertex_cycles(A, n1, fangxiang)
% GET_VERTEX_CYCLES 根据矩阵A和粘合方向计算顶点情况（保留连接顺序）
% 输入:
%   A: 粘合矩阵 (n1 x 2)
%   n1: 边数参数
%   fangxiang: 1 (同向/可定向) 或 -1 (反向/不可定向)
% 输出:
%   vertex_cells: Cell数组，每个单元包含构成该顶点的所有角点编号，且按连接顺序排列

    % 角点总数 (0 到 2*n1 - 1)
    num_corners = 2 * n1;

    % --- 1. 构建邻接表 (Adjacency List) ---
    % 使用 cell 数组存储图，adj{i} 存储与角点 i 相连的邻居
    % 索引使用 MATLAB 的 1-based，所以存取时要注意 +1
    adj = cell(num_corners, 1);

    % 内部函数: 添加无向边
    function add_edge(u, v)
        % u, v 为 0-based 索引
        idx_u = u + 1;
        idx_v = v + 1;
        adj{idx_u} = [adj{idx_u}, v];
        adj{idx_v} = [adj{idx_v}, u];
    end

    % 内部函数: 获取下一个角点 (处理周期性边界)
    function next_idx = get_next_corner(curr_idx)
        if curr_idx < n1 % 块1 (0 ~ n1-1)
            next_idx = mod(curr_idx + 1, n1);
        else % 块2 (n1 ~ 2n1-1)
            next_idx = n1 + mod(curr_idx - n1 + 1, n1);
        end
    end

    % --- 2. 填充图的连接关系 (核心逻辑修改) ---
    for i = 1:size(A, 1)
        m = A(i, 1);
        n = A(i, 2);

        m_next = get_next_corner(m);
        n_next = get_next_corner(n);
        
        % 判断是否在同一个多边形块内
        is_same_block = (m < n1 && n < n1) || (m >= n1 && n >= n1);

        if fangxiang == 1
            % 同向 (可定向): 始终是交叉粘合
            % 规则: m后继 <-> n,  m <-> n后继
            add_edge(m_next, n);
            add_edge(m, n_next);
            
        elseif fangxiang == -1
            % 反向 (不可定向): 取决于是否跨块
            if is_same_block
                % 同块内反向: 交叉粘合 (Cross Gluing)
                % 对应: [m+1, n] 和 [m, n+1]
                add_edge(m_next, n);
                add_edge(m, n_next);
            else
                % 异块间反向: 平行粘合 (Parallel Gluing)
                % 对应: [m, n] 和 [m+1, n+1]
                add_edge(m, n);
                add_edge(m_next, n_next);
            end
        end
    end

    % --- 3. 遍历图寻找回路 (串联起来) ---
    vertex_cells = {};
    visited = false(num_corners, 1); % 记录已访问的角点 (1-based)

    for i = 0 : num_corners - 1
        idx = i + 1;
        if ~visited(idx)
            % 发现新顶点，开始追踪回路
            cycle = [];
            curr = i;
            prev = -1; % 用于防止掉头回走
            
            while ~visited(curr + 1)
                visited(curr + 1) = true;
                cycle = [cycle, curr];
                
                % 找下一个未访问的邻居，或者回到起点的邻居
                neighbors = adj{curr + 1};
                found_next = false;
                
                for k = 1:length(neighbors)
                    nb = neighbors(k);
                    % 核心逻辑: 只要不是刚走过来的那个点(prev)，就是下一个点
                    if nb ~= prev
                        prev = curr;
                        curr = nb;
                        found_next = true;
                        break;
                    end
                end
                
                % 正常情况下闭曲面粘合必定闭合，且度数为2，这里作为安全出口
                if ~found_next
                    break;
                end
                
                % 如果回到了起点 (回路闭合)，停止
                if curr == i
                    break;
                end
            end
            
            vertex_cells{end+1} = cycle;
        end
    end
end

function exitflag=AVCjie(C,n1)%判断AVC是否存在一个正数解
   C_mod=cellfun(@(v) mod(v, n1), C, 'UniformOutput', false);
   num_eq = length(C_mod); % 方程数量 (行数)
num_vars = n1;          % 变量数量 (x0 到 x6，共7个变量)
target_sum = 2;
% 初始化稀疏矩阵 A (也可以用 zeros)
A = zeros(num_eq, num_vars);
b = ones(num_eq, 1) * target_sum; % 右边都是 2

for i = 1:num_eq
    indices = C_mod{i};
    % 注意：题目中是 x0, x1... (0-based)，但 MATLAB 矩阵索引从 1 开始
    % 所以 x0 对应矩阵的第 1 列，x1 对应第 2 列，以此类推。
    col_indices = indices + 1; 
    
    % 填充矩阵 A
    % 如果一个方程中同一个变量出现多次（比如 mod 后有两个 0），系数会累加
    for k = col_indices
        A(i, k) = A(i, k) + 1;
    end
end

% disp(A);
% disp('右端向量 b:');
% disp(b);

% ---------------------------------------------------------
% 步骤 3: 判断是否存在正数解 (x > 0)
% ---------------------------------------------------------
% 这是一个线性规划的可行性问题。
% 我们寻找 x 使得 Ax = b 且 x > 0。
% 标准求解器 linprog 处理的是 Ax = b, x >= lb。
% 为了表示"正数"(严格大于0)，我们将下界 lb 设为一个极小的正数 (例如 1e-6)。

lb = ones(num_vars, 1) * 1e-6; % 下界：x >= 0.000001
ub = [];                       % 上界：无无穷大
f = zeros(num_vars, 1);        % 目标函数：我们只关心可行性，设为0即可

% 求解线性规划
options = optimoptions('linprog', 'Display', 'off');
[x_sol, fval, exitflag] = linprog(f, [], [], A, b, lb, ub, options);

function edge_chains = get_edge_classes(A, n1)
% GET_EDGE_CLASSES 计算矩阵A mod n1 后的连通情况
% 输入:
%   A: 粘合矩阵
%   n1: 模数
% 输出:
%   edge_chains: Cell数组，每组包含串联起来的边索引(0 到 n1-1)

    % 1. 对矩阵取模
    A_mod = mod(A, n1);
    
    % 2. 初始化并查集
    % 范围是 0 到 n1-1，共 n1 个元素
    parent = 1:n1; 
    
    % --- 内部函数: 查找根节点 ---
    function p = find_set(x)
        path = [];
        while parent(x) ~= x
            path(end+1) = x;
            x = parent(x);
        end
        for node = path, parent(node) = x; end
        p = x;
    end

    % --- 内部函数: 合并集合 ---
    function union_set(x, y)
        rootX = find_set(x);
        rootY = find_set(y);
        if rootX ~= rootY
            parent(rootY) = rootX; % 合并
        end
    end

    % 3. 遍历每一行进行串联
    for i = 1:size(A_mod, 1)
        u = A_mod(i, 1);
        v = A_mod(i, 2);
        
        % 转为 1-based 索引进行操作
        union_set(u + 1, v + 1);
    end
    
    % 4. 整理结果 (Grouping)
    % 计算最终的根节点状态
    final_roots = arrayfun(@(x) find_set(x), 1:n1);
    unique_roots = unique(final_roots);
    
    % 仅输出包含元素的集合（理论上所有 0~n1-1 都在某个集合中）
    edge_chains = {};
    for k = 1:length(unique_roots)
        r = unique_roots(k);
        % 找出属于该组的所有边，转回 0-based
        members = find(final_roots == r) - 1;
        % 排序以便查看
        edge_chains{end+1} = sort(members);
    end
end

function [is_invalid, error_msg] = check_subset_inclusion(Cycle_Cells, n1)
% CHECK_SUBSET_INCLUSION (MOD N1 版本)
% 1. 将所有顶点对 n1 取模。
% 2. 检查是否存在 Cycle A 的变量集合 包含 Cycle B 的变量集合。
%    如果包含，且有剩余（真子集），则剩余部分之和必须为0，导致矛盾。

    num_cycles = length(Cycle_Cells);
    % 【修改点1】变量数现在只有 n1 个，不再是 2*n1
    num_vars = n1; 
    
    % 1. 构建计数矩阵 M (行=循环，列=模n1后的顶点出现次数)
    M = zeros(num_cycles, num_vars);
    for i = 1:num_cycles
        raw_data = Cycle_Cells{i};
        
        % 【修改点2】先对 n1 取模，结果为 0 ~ n1-1
        data_mod = mod(raw_data, n1);
        
        % 转为 1-based索引用于累加 (1 ~ n1)
        idx = data_mod + 1;
        
        % 统计每个 mod 后顶点的出现次数
        M(i, :) = accumarray(idx(:), 1, [num_vars, 1])';
    end
    
    is_invalid = false;
    error_msg = '';

    % 2. 暴力两两比较
    for i = 1:num_cycles       % "大"循环候选
        for j = 1:num_cycles   % "小"循环候选
            if i == j, continue; end 
            
            % 计算差值向量
            diff_vec = M(i, :) - M(j, :);
            
            % 判断逻辑：
            % 1. diff_vec >= 0: i 包含了 j 所有的变量
            % 2. any(diff_vec > 0): i 比 j 多出了至少一个变量 (真子集)
            if all(diff_vec >= 0) && any(diff_vec > 0)
                is_invalid = true;
                error_msg = sprintf('发现矛盾(Mod %d): Cycle %d 包含了 Cycle %d。', n1, i, j);
                return;
            end
        end
    end
end
